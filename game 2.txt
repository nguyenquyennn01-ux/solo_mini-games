      function normalizeList(rawObj) {
        if (!rawObj) return [];
        const arr = Array.isArray(rawObj) ? rawObj : Object.keys(rawObj).map(function (k) { return rawObj[k]; });
        return arr
          .filter(function (item) { return item && typeof item === "object" && item.name; })
          .map(function (item, idx) {
            const base = makeAccount(item.id || ("acc_" + idx), String(item.name), item.passwordHash || "");
            return Object.assign(base, item);
          });

      async function hashPassword(plainText) {
        if (!plainText) return "";
        if (window.crypto && window.crypto.subtle && window.TextEncoder) {
          const buf = new TextEncoder().encode("mgh_salt_v2:" + plainText);
          const digest = await window.crypto.subtle.digest("SHA-256", buf);
          return Array.from(new Uint8Array(digest)).map(function (b) {
            return b.toString(16).padStart(2, "0");
          }).join("");
        }
        let h = 2166136261;
        for (let i = 0; i < plainText.length; i++) {
          h ^= plainText.charCodeAt(i);
          h = Math.imul(h, 16777619);
        }
        return "fb_" + (h >>> 0).toString(16);
      }

      function spawnParticles(list, x, y, color, count) {
        for (let i = 0; i < count; i++) {
          const angle = (Math.PI * 2 * i) / count + (Math.random() - 0.5) * 0.5;
          const speed = 1.5 + Math.random() * 3.2;
          list.push({ x: x, y: y, vx: Math.cos(angle) * speed, vy: Math.sin(angle) * speed, life: 1, decay: 0.032 + Math.random() * 0.025, radius: 2 + Math.random() * 2.5, color: color });
        }
      }

      function updateAndDrawParticles(ctx, list) {
        for (let i = list.length - 1; i >= 0; i--) {
          const p = list[i];
          p.x += p.vx;
          p.y += p.vy;
          p.vx *= 0.96;
          p.vy *= 0.96;
          p.life -= p.decay;
          if (p.life <= 0) { list.splice(i, 1); continue; }
          ctx.save();
          ctx.globalAlpha = Math.max(0, p.life);
          ctx.fillStyle = p.color;
          ctx.beginPath();
          ctx.arc(p.x, p.y, p.radius, 0, Math.PI * 2);
          ctx.fill();
          ctx.restore();
        }
      }

      function getGridMetrics(n) {
        const boardPx = Math.min(pointersCanvas.width - 28, pointersCanvas.height - 28);
        const gap = n >= 7 ? 2 : 3;
        const cellSize = Math.floor((boardPx - gap * (n - 1)) / n);
        const totalSize = cellSize * n + gap * (n - 1);
        const offsetX = Math.floor((pointersCanvas.width - totalSize) / 2);
        const offsetY = Math.floor((pointersCanvas.height - totalSize) / 2);
        return { n: n, cellSize: cellSize, gap: gap, offsetX: offsetX, offsetY: offsetY, totalSize: totalSize };
      }

      function isPathClear(tilesList, c, r, dir, n) {
        let nc = c + dir.dx;
        let nr = r + dir.dy;
        while (nc >= 0 && nc < n && nr >= 0 && nr < n) {
          for (let i = 0; i < tilesList.length; i++) {
            if (tilesList[i].c === nc && tilesList[i].r === nr) {
              return tilesList[i];
            }
          }
          nc += dir.dx;
          nr += dir.dy;
        }
        return null;
      }

      function hasAnyClearTile(tilesList, n) {
        for (let i = 0; i < tilesList.length; i++) {
          const t = tilesList[i];
          if (isPathClear(tilesList, t.c, t.r, t.dir, n) === null) return true;
        }
        return false;
      }

      function distanceToEdge(c, r, dir, n) {
        if (dir.key === "RIGHT") return (n - 1) - c;
        if (dir.key === "LEFT") return c;
        if (dir.key === "DOWN") return (n - 1) - r;
        return r;
      }

      function generateFullPointersBoard(n) {
        const placed = [];
        for (let r = 0; r < n; r++) {
          for (let c = 0; c < n; c++) {
            const candidates = [];
            for (let d = 0; d < POINTER_DIRS.length; d++) {
              const dir = POINTER_DIRS[d];
              const edgeDist = distanceToEdge(c, r, dir, n);
              const testList = placed.concat([{ c: c, r: r, dir: dir }]);
              const clearNow = isPathClear(placed, c, r, dir, n) === null;
              const stillSolvable = hasAnyClearTile(testList, n);
              if (!stillSolvable && !clearNow) continue;

              let blocksExisting = 0;
              for (let k = 0; k < placed.length; k++) {
                const ex = placed[k];
                let tc = ex.c + ex.dir.dx;
                let tr = ex.r + ex.dir.dy;
                while (tc >= 0 && tc < n && tr >= 0 && tr < n) {
                  if (tc === c && tr === r) { blocksExisting++; break; }
                  tc += ex.dir.dx;
                  tr += ex.dir.dy;
                }
              }
              const weight = 1 + blocksExisting * 7 + edgeDist * 1.3;
              candidates.push({ dir: dir, weight: weight });
            }

            if (candidates.length === 0) {
              candidates.push({ dir: POINTER_DIRS[Math.floor(Math.random() * POINTER_DIRS.length)], weight: 1 });
            }

            let totalWeight = 0;
            for (let i = 0; i < candidates.length; i++) totalWeight += candidates[i].weight;
            let pick = Math.random() * totalWeight;
            let chosenDir = candidates[0].dir;
            for (let i = 0; i < candidates.length; i++) {
              pick -= candidates[i].weight;
              if (pick <= 0) { chosenDir = candidates[i].dir; break; }
            }

            placed.push({
              c: c,
              r: r,
              dir: chosenDir,
              bumpX: 0,
              bumpY: 0,
              errorFlash: 0
            });
          }
        }
        return placed;
      }

      function findTileAtCanvasPoint(px, py) {
        const m = getGridMetrics(pointerGridSize);
        for (let i = 0; i < pointerTiles.length; i++) {
          const t = pointerTiles[i];
          const x = m.offsetX + t.c * (m.cellSize + m.gap);
          const y = m.offsetY + t.r * (m.cellSize + m.gap);
          if (px >= x && px <= x + m.cellSize && py >= y && py <= y + m.cellSize) {
            return { tile: t, index: i, x: x, y: y, cellSize: m.cellSize };
          }
        }
        return null;

      function drawArrowGlyph(ctx, cx, cy, size, dir, color) {
        ctx.save();
        ctx.translate(cx, cy);
        let angle = 0;
        if (dir.key === "RIGHT") angle = 0;
        else if (dir.key === "DOWN") angle = Math.PI / 2;
        else if (dir.key === "LEFT") angle = Math.PI;
        else if (dir.key === "UP") angle = -Math.PI / 2;
        ctx.rotate(angle);

        const shaftLen = size * 0.25;
        const headSize = size * 0.22;
        ctx.strokeStyle = color;
        ctx.fillStyle = color;
        ctx.lineWidth = Math.max(2, size * 0.08);
        ctx.lineCap = "round";
        ctx.lineJoin = "round";

        ctx.beginPath();
        ctx.moveTo(-shaftLen, 0);
        ctx.lineTo(shaftLen * 0.65, 0);
        ctx.stroke();

        ctx.beginPath();
        ctx.moveTo(shaftLen + headSize * 0.45, 0);
        ctx.lineTo(shaftLen * 0.2, -headSize * 0.72);
        ctx.lineTo(shaftLen * 0.2, headSize * 0.72);
        ctx.closePath();
        ctx.fill();

        ctx.restore();
      }

      function updateAndDrawPointers(now) {
        pCtx.fillStyle = "#060913";
        pCtx.fillRect(0, 0, pointersCanvas.width, pointersCanvas.height);

        const m = getGridMetrics(pointerGridSize);

        for (let i = 0; i < pointerTiles.length; i++) {
          const t = pointerTiles[i];
          t.bumpX *= 0.74;
          t.bumpY *= 0.74;
          if (t.errorFlash > 0) t.errorFlash = Math.max(0, t.errorFlash - 0.055);

          const key = t.c + "," + t.r;
          const isHovered = hoveredTileKey === key && pointersRunning;
          const isHinted = hintTileKey === key && now < hintUntil;

          const x = m.offsetX + t.c * (m.cellSize + m.gap) + t.bumpX;
          const y = m.offsetY + t.r * (m.cellSize + m.gap) + t.bumpY;

          let arrowColor = "#e2e8f0";
          if (t.errorFlash > 0) arrowColor = "#ef4444";
          else if (isHinted) arrowColor = "#4ade80";
          else if (isHovered) arrowColor = "#38bdf8";

          drawArrowGlyph(
            pCtx,
            x + m.cellSize / 2,
            y + m.cellSize / 2,
            m.cellSize,
            t.dir,
            arrowColor
          );
        }

        for (let i = exitingTiles.length - 1; i >= 0; i--) {
          const ex = exitingTiles[i];
          ex.x += ex.vx;
          ex.y += ex.vy;
          ex.vx *= 1.08;
          ex.vy *= 1.08;

            ex.x < -ex.size - 20 ||
            ex.x > pointersCanvas.width + 20 ||
            ex.y < -ex.size - 20 ||
            ex.y > pointersCanvas.height + 20

            exitingTiles.splice(i, 1);
            continue;
          }

          drawArrowGlyph(pCtx, ex.x + ex.size / 2, ex.y + ex.size / 2, ex.size, ex.dir, "#38bdf8");
        }

        updateAndDrawParticles(pCtx, pointerParticles);
      }

      function cloneCars(list) {
        return list.map(function (c) {
          return {
            id: c.id,
            c: c.c,
            r: c.r,
            len: c.len,
            orient: c.orient,
            isRed: Boolean(c.isRed),
            color: c.color
          };
        });
      }

      function buildOccupancy(cars, ignoreId) {
        const occ = Array.from({ length: SC_GRID }, function () {
          return Array(SC_GRID).fill(null);
        });
        for (let i = 0; i < cars.length; i++) {
          const car = cars[i];
          if (ignoreId && car.id === ignoreId) continue;
          for (let k = 0; k < car.len; k++) {
            const cc = car.orient === "H" ? car.c + k : car.c;
            const rr = car.orient === "V" ? car.r + k : car.r;
            if (cc >= 0 && cc < SC_GRID && rr >= 0 && rr < SC_GRID) {
              occ[rr][cc] = car.id;
            }
          }
        }
        return occ;
      }

      function computeMoveBounds(car, cars) {
        const occ = buildOccupancy(cars, car.id);
        if (car.orient === "H") {
          let minC = car.c;
          while (minC - 1 >= 0 && occ[car.r][minC - 1] === null) minC--;
          let maxC = car.c;
          while (maxC + car.len < SC_GRID && occ[car.r][maxC + car.len] === null) maxC++;
          return { min: minC, max: maxC };
        } else {
          let minR = car.r;
          while (minR - 1 >= 0 && occ[minR - 1][car.c] === null) minR--;
          let maxR = car.r;
          while (maxR + car.len < SC_GRID && occ[maxR + car.len][car.c] === null) maxR++;
          return { min: minR, max: maxR };
        }
      }

      function loadSlideCarsLevel(lvlNum) {
        const cfg = DIFF_CONFIG[currentDiff];
        const tpl = SLIDE_CAR_LEVELS[(lvlNum - 1) % SLIDE_CAR_LEVELS.length];
        const extraBlock = Math.floor((lvlNum - 1) / SLIDE_CAR_LEVELS.length);
        const copied = cloneCars(tpl.cars);
        if (extraBlock > 0) {
          const occ = buildOccupancy(copied, null);
          for (let r = 0; r < SC_GRID; r++) {
            if (r === EXIT_ROW) continue;
            if (occ[r][4] === null && occ[r][5] === null) {
              copied.push({
                id: "extra_" + lvlNum + "_" + r,
                c: 4,
                r: r,
                len: 2,
                orient: "H",
                isRed: false,
                color: "#64748b"
              });
              break;
            }
          }
        }
        slideCarsInitial = cloneCars(copied);
        slideCarsList = cloneCars(copied);
        slideCarsMovesLeft = tpl.parMoves + cfg.slideMovesBonus + (lvlNum - 1) * 2;
        slideCarsLevelStartScore = slideCarsScore;
        slideCarsExiting = false;
        draggedCar = null;
        updateSlideCarsHUD();
        syncLockVisuals();
        slideCarsStatusEl.textContent = "Màn " + lvlNum + " (" + cfg.label + "): Trượt xe mở lối và đưa xe đỏ ra cổng phải!";
      }

      function getSlideCarsCoords(clientX, clientY) {
        const rect = slideCarsCanvas.getBoundingClientRect();
        return {
          x: ((clientX - rect.left) / rect.width) * slideCarsCanvas.width,
          y: ((clientY - rect.top) / rect.height) * slideCarsCanvas.height
        };
      }

      function getCarRect(car) {
        const x = SC_OFFSET_X + car.c * (SC_CELL + SC_GAP);
        const y = SC_OFFSET_Y + car.r * (SC_CELL + SC_GAP);
        const w = car.orient === "H" ? car.len * SC_CELL + (car.len - 1) * SC_GAP : SC_CELL;
        const h = car.orient === "V" ? car.len * SC_CELL + (car.len - 1) * SC_GAP : SC_CELL;
        return { x: x, y: y, w: w, h: h };
      }

      slideCarsCanvas.addEventListener("pointerdown", function (e) {
        if (!slideCarsRunning || slideCarsExiting || activeGame !== "slidecars") return;
        const pt = getSlideCarsCoords(e.clientX, e.clientY);
        for (let i = 0; i < slideCarsList.length; i++) {
          const car = slideCarsList[i];
          const rect = getCarRect(car);
          if (pt.x >= rect.x && pt.x <= rect.x + rect.w && pt.y >= rect.y && pt.y <= rect.y + rect.h) {
            const bounds = computeMoveBounds(car, slideCarsList);
            draggedCar = {
              car: car,
              min: bounds.min,
              max: bounds.max
            };
            dragStartPt = pt;
            dragStartCell = car.orient === "H" ? car.c : car.r;
            return;
          }
        }
      });

      slideCarsCanvas.addEventListener("pointermove", function (e) {
        if (!draggedCar || !slideCarsRunning || slideCarsExiting) return;
        const pt = getSlideCarsCoords(e.clientX, e.clientY);
        const stepPx = SC_CELL + SC_GAP;
        const delta = draggedCar.car.orient === "H"
          ? (pt.x - dragStartPt.x) / stepPx
          : (pt.y - dragStartPt.y) / stepPx;
        const nextPos = Math.max(draggedCar.min, Math.min(draggedCar.max, Math.round(dragStartCell + delta)));
        if (draggedCar.car.orient === "H") draggedCar.car.c = nextPos;
        else draggedCar.car.r = nextPos;
      });

      function finishCarSlide() {
        if (!draggedCar) return;
        const car = draggedCar.car;
        const prevPos = dragStartCell;
        const newPos = car.orient === "H" ? car.c : car.r;
        draggedCar = null;

        if (newPos === prevPos) return;

        slideCarsMovesLeft--;
        updateSlideCarsHUD();

        if (car.isRed && car.c + car.len === SC_GRID) {
          slideCarsExiting = true;
          redCarExitX = SC_OFFSET_X + car.c * (SC_CELL + SC_GAP);
          const bonus = 50 + Math.max(0, slideCarsMovesLeft) * 10 + slideCarsLevel * 20;
          slideCarsScore += bonus;
          if (slideCarsScore > slideCarsBest) {
            slideCarsBest = slideCarsScore;
          }
          saveBestOnly("slideCarsBest", slideCarsScore);
          updateSlideCarsHUD();
          spawnParticles(slideCarsParticles, SC_OFFSET_X + SC_BOARD_PX + 10, SC_OFFSET_Y + EXIT_ROW * (SC_CELL + SC_GAP) + SC_CELL / 2, "#ef4444", 18);
          const cleared = slideCarsLevel;
          slideCarsLevel++;
          slideCarsStatusEl.innerHTML = '<span class="win">🚗 Xe đỏ đã thoát ra ở Màn ' + cleared + " (+" + bonus + " điểm)! Đang sang Màn " + slideCarsLevel + "...</span>";
          setTimeout(function () {
            if (slideCarsRunning) {
              loadSlideCarsLevel(slideCarsLevel);
            }
          }, 800);
          return;
        }

        if (slideCarsMovesLeft <= 0) {
          slideCarsRunning = false;
          syncLockVisuals();
          slideCarsStatusEl.innerHTML = '<span class="lose">⛔ Hết lượt trượt ở Màn ' + slideCarsLevel + "! Điểm của bạn: " + slideCarsScore + ".</span>";
          updateResult(slideCarsScore > 0 ? "win" : "loss", "slideCarsBest", slideCarsScore, null);
        }
      }

      function drawDetailedCar(ctx, x, y, w, h, car, isDragging, pulsePhase) {
        const pad = 2.5;
        const bx = x + pad;
        const by = y + pad;
        const bw = w - pad * 2;
        const bh = h - pad * 2;
        const radius = 11;

        ctx.save();

        // Bánh xe nhô nhẹ ở hai đầu trục (làm rõ dáng ô tô)
        ctx.fillStyle = car.isRed ? "#111827" : "#090d16";
        ctx.strokeStyle = car.isRed ? "#fda4af" : "rgba(148, 163, 184, 0.35)";
        ctx.lineWidth = car.isRed ? 1.6 : 1;
        const wheelThick = 5;
        const wheelLen = 12;

        if (car.orient === "H") {
          const leftWheelX = bx + 9;
          const rightWheelX = bx + bw - 9 - wheelLen;
          [leftWheelX, rightWheelX].forEach(function (wx) {
            ctx.beginPath();
            ctx.roundRect(wx, by - wheelThick + 2, wheelLen, wheelThick, 2);
            ctx.roundRect(wx, by + bh - 2, wheelLen, wheelThick, 2);
            ctx.fill();
            ctx.stroke();
          });
        } else {
          const topWheelY = by + 9;
          const botWheelY = by + bh - 9 - wheelLen;
          [topWheelY, botWheelY].forEach(function (wy) {
            ctx.beginPath();
            ctx.roundRect(bx - wheelThick + 2, wy, wheelThick, wheelLen, 2);
            ctx.roundRect(bx + bw - 2, wy, wheelThick, wheelLen, 2);
            ctx.fill();
            ctx.stroke();
          });
        }

        if (car.isRed) {
          const glowAlpha = 0.45 + Math.sin(pulsePhase) * 0.25;
          ctx.shadowColor = "rgba(251, 113, 133, " + glowAlpha.toFixed(2) + ")";
          ctx.shadowBlur = isDragging ? 20 : 14;
        } else if (isDragging) {
          ctx.shadowColor = "rgba(56, 189, 248, 0.45)";
          ctx.shadowBlur = 12;
        }

        ctx.fillStyle = car.isRed ? "#ef4444" : car.color;
        ctx.beginPath();
        ctx.roundRect(bx, by, bw, bh, radius);
        ctx.fill();

        ctx.shadowBlur = 0;

        if (car.isRed) {
          ctx.strokeStyle = "#ffffff";
          ctx.lineWidth = 2.6;
        } else if (isDragging) {
          ctx.strokeStyle = "#f8fafc";
          ctx.lineWidth = 2.2;
        } else {
          ctx.strokeStyle = "rgba(248, 250, 252, 0.35)";
          ctx.lineWidth = 1.4;
        }
        ctx.beginPath();
        ctx.roundRect(bx, by, bw, bh, radius);
        ctx.stroke();

        if (car.orient === "H") {
          const cabinX = bx + bw * 0.22;
          const cabinW = bw * 0.54;
          const cabinY = by + 7;
          const cabinH = bh - 14;

          ctx.fillStyle = car.isRed ? "rgba(15, 23, 42, 0.72)" : "rgba(6, 9, 19, 0.42)";
          ctx.beginPath();
          ctx.roundRect(cabinX, cabinY, cabinW, cabinH, 6);
          ctx.fill();

          ctx.fillStyle = car.isRed ? "rgba(186, 230, 253, 0.68)" : "rgba(226, 232, 240, 0.24)";
          ctx.beginPath();
          ctx.roundRect(cabinX + cabinW - 9, cabinY + 3, 6, Math.max(4, cabinH - 6), 2);
          ctx.roundRect(cabinX + 3, cabinY + 3, 5, Math.max(4, cabinH - 6), 2);
          ctx.fill();

          if (car.isRed) {
            ctx.fillStyle = "#fef08a";
            ctx.beginPath();
            ctx.arc(bx + bw - 5, by + 11, 3.3, 0, Math.PI * 2);
            ctx.arc(bx + bw - 5, by + bh - 11, 3.3, 0, Math.PI * 2);
            ctx.fill();

            ctx.fillStyle = "#ffffff";
            ctx.font = "900 12px system-ui, sans-serif";
            ctx.textAlign = "center";
            ctx.textBaseline = "middle";
            ctx.fillText("➔", bx + bw * 0.5, by + bh / 2 + 0.5);
          }
        } else {
          const cabinX = bx + 7;
          const cabinW = bw - 14;
          const cabinY = by + bh * 0.22;
          const cabinH = bh * 0.54;

          ctx.fillStyle = "rgba(6, 9, 19, 0.42)";
          ctx.beginPath();
          ctx.roundRect(cabinX, cabinY, cabinW, cabinH, 6);

          ctx.roundRect(cabinX + 3, cabinY + cabinH - 8, Math.max(4, cabinW - 6), 5, 2);
          ctx.fill();
        }

        ctx.restore();
      }

      const cells = document.querySelectorAll(".ttt-cell");

      const winPatterns = [[0,1,2],[3,4,5],[6,7,8],[0,3,6],[1,4,7],[2,5,8],[0,4,8],[2,4,6]];

      function findWinLine(grid) {
        for (let i = 0; i < winPatterns.length; i++) {
          const line = winPatterns[i];
          const a = line[0], b = line[1], c = line[2];
          if (grid[a] && grid[a] === grid[b] && grid[a] === grid[c]) return { winner: grid[a], line: line };
        }
        return grid.indexOf("") === -1 ? { winner: "draw", line: null } : null;
      }

      function evaluateBoardHeuristic(grid) {
        let score = 0;
        for (let i = 0; i < winPatterns.length; i++) {
          const line = winPatterns[i];
          let oCount = 0;
          let xCount = 0;
          for (let j = 0; j < 3; j++) {
            const mark = grid[line[j]];
            if (mark === "O") oCount++;
            else if (mark === "X") xCount++;
          }
          if (oCount > 0 && xCount === 0) score += oCount * oCount * 10;
          else if (xCount > 0 && oCount === 0) score -= xCount * xCount * 10;
        }
        if (grid[4] === "O") score += 6;
        else if (grid[4] === "X") score -= 6;
        return score;
      }

      function minimax(grid, depth, maxDepth, isMaximizing, alpha, beta) {
        const res = findWinLine(grid);
        if (res) {
          if (res.winner === "O") return 1000 + depth;
          if (res.winner === "X") return -1000 - depth;
          return 0;
        }
        if (depth >= maxDepth) {
          return evaluateBoardHeuristic(grid);
        }

        if (isMaximizing) {
          let best = -Infinity;
          for (let i = 0; i < 9; i++) {
            if (!grid[i]) {
              grid[i] = "O";
              const val = minimax(grid, depth + 1, maxDepth, false, alpha, beta);
              grid[i] = "";
              if (val > best) best = val;
              if (best > alpha) alpha = best;
              if (beta <= alpha) break;
            }
          }
          return best;
        } else {
          let best = Infinity;
          for (let i = 0; i < 9; i++) {
            if (!grid[i]) {
              grid[i] = "X";
              const val = minimax(grid, depth + 1, maxDepth, true, alpha, beta);
              grid[i] = "";
              if (val < best) best = val;
              if (best < beta) beta = best;
              if (beta <= alpha) break;
            }
          }
          return best;
        }
      }

      function chooseAIMove() {
        const cfg = DIFF_CONFIG[currentDiff];
        const empty = [];
        for (let i = 0; i < 9; i++) {
          if (!tttBoard[i]) empty.push(i);
        }
        if (empty.length === 0) return -1;
        if (cfg.tttRandomChance > 0 && Math.random() < cfg.tttRandomChance) {
          return empty[Math.floor(Math.random() * empty.length)];
        }

        let bestScore = -Infinity;
        let bestMoves = [];
        for (let idx = 0; idx < empty.length; idx++) {
          const move = empty[idx];
          tttBoard[move] = "O";
          const score = minimax(tttBoard, 0, cfg.tttDepth, false, -Infinity, Infinity);
          tttBoard[move] = "";
          if (score > bestScore) {
            bestScore = score;
            bestMoves = [move];
          } else if (score === bestScore) {
            bestMoves.push(move);
          }
        }
        return bestMoves[Math.floor(Math.random() * bestMoves.length)];
      }

        const lastPlanned = dirQueue.length > 0 ? dirQueue[dirQueue.length - 1] : snakeDir;
        const isOpposite = lastPlanned.x === -newDir.x && lastPlanned.y === -newDir.y;

        if (!isOpposite && !isSame && dirQueue.length < 2) dirQueue.push(newDir);

      function randomFood() {
        let pos;
        do { pos = { x: Math.floor(Math.random() * tileCount), y: Math.floor(Math.random() * tileCount) }; }
        while (snake.some(function (seg) { return seg.x === pos.x && seg.y === pos.y; }));

        sCtx.fillStyle = "#060913";

        sCtx.strokeStyle = "rgba(148, 163, 184, 0.06)";
        sCtx.lineWidth = 1;
        for (let i = 1; i < tileCount; i++) {
          sCtx.beginPath(); sCtx.moveTo(i * gridSize, 0); sCtx.lineTo(i * gridSize, snakeCanvas.height); sCtx.stroke();

        snakeRunning = false;

      function stepSnake() {
        if (!snakeRunning) return;
        if (dirQueue.length > 0) snakeDir = dirQueue.shift();
        const head = { x: snake[0].x + snakeDir.x, y: snake[0].y + snakeDir.y };
        if (head.x < 0 || head.x >= tileCount || head.y < 0 || head.y >= tileCount) { endSnake("Đâm vào tường rồi!"); return; }
        if (snake.some(function (seg) { return seg.x === head.x && seg.y === head.y; })) { endSnake("Rắn cắn trúng thân mình!"); return; }
        snake.unshift(head);
        if (head.x === food.x && head.y === food.y) {
          spawnParticles(snakeParticles, food.x * gridSize + gridSize / 2, food.y * gridSize + gridSize / 2, "#fb7185", 12);
          snakeScore += 10; snakeScoreEl.textContent = String(snakeScore);
          if (snakeScore > snakeBest) { snakeBest = snakeScore; snakeBestEl.textContent = String(snakeBest); }
          food = randomFood();
        } else { snake.pop(); }
      }

            setFeedback("Đang chơi trò chơi này — không được đổi sang trò chơi khác! Hãy hoàn thành hoặc đặt lại ván trước.", "err");
            return;
          }
          activeGame = targetGame;
          tabs.forEach(function (t) { t.classList.remove("active"); });
          sections.forEach(function (s) { s.classList.remove("active"); });
          tab.classList.add("active");
          document.getElementById(activeGame).classList.add("active");
          syncLockVisuals();
        });
      });

      window.addEventListener("keydown", function (e) {
        if (e.key === "Escape") {
          document.querySelectorAll(".modal-backdrop.open").forEach(function (m) { m.classList.remove("open"); });
          return;
        }
        if (document.querySelector(".modal-backdrop.open")) return;
        if (activeGame === "snake") {
          if (e.code === "ArrowUp" || e.code === "KeyW") queueSnakeDir({ x: 0, y: -1 });
          if (e.code === "ArrowDown" || e.code === "KeyS") queueSnakeDir({ x: 0, y: 1 });
          if (e.code === "ArrowLeft" || e.code === "KeyA") queueSnakeDir({ x: -1, y: 0 });
          if (e.code === "ArrowRight" || e.code === "KeyD") queueSnakeDir({ x: 1, y: 0 });
          if (["ArrowUp", "ArrowDown", "ArrowLeft", "ArrowRight"].indexOf(e.code) !== -1) e.preventDefault();
        }
        if (activeGame === "runner" && (e.code === "Space" || e.code === "ArrowUp" || e.code === "KeyW")) {
          e.preventDefault(); triggerJump();
        }
      });

      renderProfileUI();
      applyDifficultyToUI();
      syncFromCloud();
    })();
  </script>
